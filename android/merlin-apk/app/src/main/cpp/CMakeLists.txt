cmake_minimum_required(VERSION 3.22)
project(merlin_android_local_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp/include/llama.h")
  message(FATAL_ERROR "Pinned llama.cpp missing: run scripts/merlin/prepare_android_native.py")
endif()
add_subdirectory(vendor/llama.cpp "${CMAKE_CURRENT_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
add_library(merlin_local_ai SHARED merlin_model_jni.cpp)
target_include_directories(merlin_local_ai PRIVATE vendor/llama.cpp/include)
target_link_libraries(merlin_local_ai PRIVATE llama)
target_compile_options(merlin_local_ai PRIVATE -Wall -Wextra -Werror=return-type)
